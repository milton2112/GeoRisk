      try {
        await context.addInitScript(() => {
          Object.defineProperty(navigator, 'hardwareConcurrency', { get: () => 4 });
          Object.defineProperty(navigator, 'deviceMemory', { get: () => 16 });
          localStorage.setItem('geo-risk-intro-seen', 'true');
          localStorage.setItem('geo-risk-auto-rotate', 'false');
          localStorage.setItem('geo-risk-quality-preset', 'auto');
        });
        const page = await context.newPage();
        page.on('pageerror', error => { if (result.errors.length < 12) result.errors.push(error.message); });
        const cdp = await context.newCDPSession(page);
        await cdp.send('Emulation.setCPUThrottlingRate', { rate: mobile ? 4 : 1 });
        await page.goto(`http://127.0.0.1:${server.address().port}/index.html`, { waitUntil: 'domcontentloaded', timeout: 45000 });
        await page.waitForFunction(() => typeof viewer !== 'undefined' && viewer && !document.body.classList.contains('globe-loading') && countryLayers.has('ARG') && countryLayers.has('ESP') && viewer.scene.globe.tilesLoaded && !isCameraNavigating && !loadMapPromise, undefined, { timeout: 45000 });
        await page.evaluate(() => document.fonts.ready);
        await page.waitForFunction(() => activeImagerySignature === currentMapMode + (isMobileLayout() ? ':boot' : ':full') && !isCameraNavigating && navigationQualityRestoreTimer === null && viewer.scene.globe.tilesLoaded, undefined, { timeout: 45000 });
        await page.evaluate(() => {
          qualityPreset = 'balanced'; // Stop automatic adaptation in this pixel-only control; preserve actual initial AA/profile.
          const gl = viewer.scene.context._gl;
          const native = gl.fenceSync;
          const own = Object.getOwnPropertyDescriptor(gl, 'fenceSync');
          const state = window.__pixelResizeProbe = { enabled: false, created: 0, bypassed: 0 };
          const wrapped = function (...args) {
            if (!state.enabled) { state.bypassed++; return null; }
            state.created++; return native.apply(this, args);
          };
          gl.fenceSync = wrapped;
          state.restore = () => { if (gl.fenceSync === wrapped) { if (own) Object.defineProperty(gl, 'fenceSync', own); else delete gl.fenceSync; } };
        });
        const buffers = [];
        for (const [index, enabled] of [false, true, false].entries()) {
          const capture = await page.evaluate(async ({ enabled, mobile }) => {
            const scene = viewer.scene;
            const widget = viewer.cesiumWidget;
            const gl = scene.context._gl;
            const probe = window.__pixelResizeProbe;
            probe.enabled = enabled;
            const before = { created: probe.created, bypassed: probe.bypassed };
            const target = mobile ? 0.44 : 0.94;
            const wait = (scale, pixels = false) => new Promise((resolve, reject) => {
              let stable = 0;
              const remove = scene.postRender.addEventListener(() => {
                if (widget._forceResize || widget._geoRiskResizeFence || !scene.globe.tilesLoaded || isCameraNavigating || navigationQualityRestoreTimer !== null ||
                    scene.canvas.width !== Math.floor(scene.canvas.clientWidth * scale) || scene.canvas.height !== Math.floor(scene.canvas.clientHeight * scale)) stable = 0;
                else stable++;
                if (stable < 3) { scene.requestRender(); return; }
                remove(); clearTimeout(timer);
                if (!pixels) { resolve(); return; }
                const bytes = new Uint8Array(scene.canvas.width * scene.canvas.height * 4);
                if (bytes.length > 6 * 1024 ** 2) { reject(new Error('Pixel bound exceeded')); return; }
                gl.readPixels(0, 0, scene.canvas.width, scene.canvas.height, gl.RGBA, gl.UNSIGNED_BYTE, bytes);
                let binary = '';
                for (let i = 0; i < bytes.length; i += 8192) binary += String.fromCharCode(...bytes.subarray(i, i + 8192));
                resolve({ pixels: btoa(binary), width: scene.canvas.width, height: scene.canvas.height, resolution: viewer.resolutionScale,
                  mode: currentMapMode, msaa: scene.msaaSamples, fxaa: scene.postProcessStages.fxaa.enabled, imagery: activeImagerySignature,
                  labels: labelEntities.length, pending: Boolean(widget._geoRiskResizeFence), navigationQualityRestoreTimer,
                  globeError: scene.globe.maximumScreenSpaceError, tileCache: scene.globe.tileCacheSize,
                  position: Cesium.Cartesian3.clone(viewer.camera.positionWC), direction: Cesium.Cartesian3.clone(viewer.camera.directionWC),
                  readFramebufferDefault: gl.getParameter(gl.READ_FRAMEBUFFER_BINDING) === null, glError: gl.getError() });
              });
              const timer = setTimeout(() => { remove(); reject(new Error('Native resize/frame capture timeout')); }, 15000);
              scene.requestRender();
            });
            viewer.resolutionScale = mobile ? 0.5 : 0.98;
            await wait(viewer.resolutionScale);
            viewer.resolutionScale = target;
            await wait(target);
            fitWorldView();
            const result = await wait(target, true);
            return { ...result, enabled, created: probe.created - before.created, bypassed: probe.bypassed - before.bypassed };
          }, { enabled, mobile });
          const bytes = Buffer.from(capture.pixels, 'base64');
          delete capture.pixels;
          assert.equal(bytes.length, capture.width * capture.height * 4);
          assert.ok(new Set(bytes).size > 32, 'nonblank rendered image');
          buffers.push(bytes);
          result.captures.push({ ...capture, rgbaBytes: bytes.length, sha256: createHash('sha256').update(bytes).digest('hex') });
          assert.equal(capture.glError, 0);
          assert.equal(capture.pending, false);
          assert.ok(enabled ? capture.created > 0 && capture.bypassed === 0 : capture.bypassed > 0 && capture.created === 0, 'exercise actual native guard/fallback');
          await page.screenshot({ path: path.join(output, `${name}-${index}.png`) });
        }
        await page.evaluate(() => window.__pixelResizeProbe.restore());
        result.controlEqual = buffers[0].equals(buffers[2]);
        result.guardEqual = buffers[0].equals(buffers[1]);
        result.pixelDifferences = [1,2].map(index => {
          let pixels = 0, maxChannel = 0;
          for (let i = 0; i < buffers[0].length; i += 4) {
            let changed = false;
            for (let channel = 0; channel < 4; channel++) {
              const delta = Math.abs(buffers[0][i + channel] - buffers[index][i + channel]);
              maxChannel = Math.max(maxChannel, delta); changed ||= delta !== 0;
            }
            if (changed) pixels++;
          }
          return { index, pixels, maxChannel };
        });
        if (!result.controlEqual || !result.guardEqual) {
          for (let index = 0; index < buffers.length; index++) {
            const { width, height } = result.captures[index];
            const image = new PNG({ width, height });
            for (let row = 0; row < height; row++) buffers[index].copy(image.data, (height - 1 - row) * width * 4, row * width * 4, (row + 1) * width * 4);
            await fs.writeFile(path.join(output, `${name}-${index}-raw.png`), PNG.sync.write(image));
          }
        }
        assert.equal(result.controlEqual, true, 'fallback/fallback pixels stable');
        assert.equal(result.guardEqual, true, 'guard preserves every RGBA byte and dimensions');
        assert.deepEqual(result.errors, []);
        result.passed = true;
        console.log(JSON.stringify({ name, controlEqual: result.controlEqual, guardEqual: result.guardEqual, captures: result.captures.map(({ pixels, ...capture }) => capture), errors: result.errors }));
      } catch (error) { result.failure = error.message; throw error; }
      finally { await context.close(); }
    }
    await browser.close(); browser = null;
  }
  report.passed = true;
} catch (error) { report.failure = error.message; throw error; }
finally {
  report.finishedAt = new Date().toISOString();
  await fs.writeFile(path.join(output, 'canvas-resize-pixels.json'), JSON.stringify(report, null, 2) + '\n');
  await browser?.close();
  await new Promise(resolve => server.close(resolve));
}
