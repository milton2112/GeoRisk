export function browserFlowConcurrency(value = "1") {
  if (value !== "1" && value !== "2") throw new Error("Browser flow concurrency must be 1 or 2.");
  return Number(value);
}

// The caller selects independent flows; map/performance/shared-server flows stay serial.
export async function runBrowserFlowBatch(flows, execute, concurrency = 1) {
  if (![1, 2].includes(concurrency)) throw new Error("Browser flow concurrency must be 1 or 2.");
  let next = 0;
  let failed = false;
  let firstError;
  async function worker(slot) {
    while (!failed && next < flows.length) {
      const flow = flows[next++];
      try { await execute(flow, slot); }
      catch (error) { if (!failed) { failed = true; firstError = error; } }
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, flows.length) }, (_, slot) => worker(slot)));
  if (failed) throw firstError;
}
